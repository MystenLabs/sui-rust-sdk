52    /// Construct a client for a faucet service URL.
53    ///
54    /// The URL must not include an API path. The service must implement the v3 faucet API.
55    pub fn new<T>(faucet_url: T) -> Result<Self, BoxError>
56    where
57        T: TryInto<http::Uri>,
58        T::Error: Into<BoxError>,
59    {
60        let inner = reqwest::Client::new();
61        let faucet_url = faucet_url.try_into().map_err(Into::into)?;
62        Ok(Self { faucet_url, inner })
63    }

65    /// Fetch a proof-of-work challenge for `recipient`.
66    pub async fn create_challenge(&self, recipient: Address) -> Result<PowChallenge, BoxError> {
67        self.fetch_challenge(recipient)
68            .await?
69            .ok_or_else(|| "faucet does not support v3 proof-of-work challenges".into())
70    }

72    /// Request faucet funds, using v3 proof of work when the faucet supports it.
73    ///
74    /// This method falls back to v2 only when `GET /v3/challenge` returns 404. It solves v3 proofs
75    /// on Tokio's blocking thread pool. Once started, solving continues even if this future is
76    /// dropped. Use [`Self::create_challenge`], [`PowChallenge::solve`], and [`Self::submit`] to
77    /// schedule that work separately.
78    pub async fn request(&self, recipient: Address) -> Result<FaucetResponse, BoxError> {
79        match self.fetch_challenge(recipient).await? {
80            Some(challenge) => {
81                let (challenge, solution) = tokio::task::spawn_blocking(move || {
82                    let solution = challenge.solve()?;
83                    Ok::<_, BoxError>((challenge, solution))
84                })
85                .await??;
86                self.submit(&challenge, &solution)
87                    .await
88                    .map(FaucetResponse::V3)
89            }
90            None => self.request_v2(recipient).await.map(FaucetResponse::V2),
91        }
92    }
93
94    /// Submit a solved proof-of-work challenge.
95    pub async fn submit(
96        &self,
97        challenge: &PowChallenge,
98        solution: &PowSolution,
99    ) -> Result<V3FaucetResponse, BoxError> {
100        challenge.validate(challenge.recipient)?;
101        let response = self
102            .inner
103            .post(self.endpoint("v3/gas"))
104            .json(&FaucetRequest {
105                recipient: challenge.recipient,
106                checkpoint_seq: challenge.checkpoint_seq.to_string(),
107                nonce: solution.nonce.to_string(),
108                hash_hex: solution.hash_hex(),
109            })
110            .send()
111            .await?;
112        self.json_response(response).await
113    }
114
115    async fn fetch_challenge(&self, recipient: Address) -> Result<Option<PowChallenge>, BoxError> {
116        let response = self
117            .inner
118            .get(self.endpoint("v3/challenge"))
119            .query(&[("recipient", recipient.to_string())])
120            .send()
121            .await?;
122        if response.status() == StatusCode::NOT_FOUND {
123            return Ok(None);
124        }
125        let challenge: PowChallenge = self.json_response(response).await?;
126        challenge.validate(recipient)?;
127        Ok(Some(challenge))
128    }
129
130    async fn request_v2(&self, recipient: Address) -> Result<Vec<CoinInfo>, BoxError> {
131        let response = self
132            .inner
133            .post(self.endpoint("v2/gas"))
134            .json(&V2FaucetRequest {
135                fixed_amount_request: V2FixedAmountRequest { recipient },
136            })
137            .send()
138            .await?;
139        let response: V2FaucetResponse = self.json_response(response).await?;
140        match response.status {
141            V2RequestStatus::Success => Ok(response.coins_sent.unwrap_or_default()),
142            V2RequestStatus::Failure(error) => {
143                Err(format!("v2 faucet request failed: {error}").into())
144            }
145        }
146    }
147
148    fn endpoint(&self, path: &str) -> String {
149        format!(
150            "{}/{path}",
151            self.faucet_url.to_string().trim_end_matches('/')
152        )
153    }
154
155    async fn json_response<T>(&self, response: reqwest::Response) -> Result<T, BoxError>
156    where
157        T: serde::de::DeserializeOwned,
158    {
159        let status = response.status();
160        if status.is_success() {
161            return Ok(response.json().await?);
162        }
163
164        let body = response
165            .json::<serde_json::Value>()
166            .await
167            .unwrap_or_default();
168        // Decode fields independently so an incompatible challenge cannot hide a payout digest.
169        Err(FaucetError {
170            status,
171            code: body.get("code").and_then(|v| v.as_str()).map(str::to_owned),
172            message: body
173                .get("error")
174                .and_then(|v| v.as_str())
175                .map(str::to_owned)
176                .unwrap_or_else(|| status.to_string()),
177            digest: body
178                .get("digest")
179                .and_then(|v| v.as_str())
180                .and_then(|v| v.parse().ok()),
181            challenge: body
182                .get("challenge")
183                .cloned()
184                .and_then(|v| serde_json::from_value(v).ok()),
185        }
186        .into())
187    }
188}

223impl PowChallenge {
224    /// Solve this challenge and return the proof accepted by the v3 faucet.
225    ///
226    /// This method is CPU and memory intensive. Call it from a blocking task when running in an
227    /// asynchronous application.
228    pub fn solve(&self) -> Result<PowSolution, BoxError> {
229        self.validate(self.recipient)?;
230        let params = Params::new(
231            POW_MEMORY_SIZE,
232            POW_ITERATIONS,
233            POW_PARALLELISM,
234            Some(POW_HASH_LENGTH),
235        )
236        .map_err(|error| format!("invalid proof-of-work parameters: {error}"))?;
237        let mut memory_blocks = vec![argon2::Block::default(); params.block_count()];
238        let argon2 = Argon2::new(Algorithm::Argon2d, Version::V0x13, params);
239
240        // Vary the start to reduce duplicate proofs for the same checkpoint and recipient.
241        let start_nonce = RandomState::new().hash_one(());
242        for offset in 0..=u64::MAX {
243            let nonce = start_nonce.wrapping_add(offset);
244            let mut hash = [0; POW_HASH_LENGTH];
245            argon2
246                .hash_password_into_with_memory(
247                    self.preimage(nonce).as_bytes(),
248                    POW_SALT,
249                    &mut hash,
250                    &mut memory_blocks,
251                )
252                .map_err(|error| format!("failed to compute proof of work: {error}"))?;
253            if hash_to_u64(&hash) < self.threshold {
254                return Ok(PowSolution { nonce, hash });
255            }
256        }
257
258        Err("exhausted the proof-of-work nonce space".into())
259    }
260
261    fn validate(&self, recipient: Address) -> Result<(), BoxError> {
262        let valid_parameters = self.version == POW_VERSION
263            && self.domain == POW_DOMAIN
264            && self.algorithm == POW_ALGORITHM
265            && self.argon2_version == POW_ARGON2_VERSION
266            && self.memory_size == POW_MEMORY_SIZE
267            && self.iterations == POW_ITERATIONS
268            && self.parallelism == POW_PARALLELISM
269            && self.hash_length == POW_HASH_LENGTH
270            && self.salt.as_bytes() == POW_SALT;
271        if !valid_parameters {
272            return Err("faucet challenge uses unsupported proof-of-work parameters".into());
273        }
274        if self.recipient != recipient {
275            return Err("faucet challenge recipient does not match the requested recipient".into());
276        }
277        if !(MIN_DIFFICULTY..=MAX_DIFFICULTY).contains(&self.difficulty) {
278            return Err("faucet challenge has an invalid difficulty".into());
279        }
280        if self.difficulty_base != self.difficulty
281            || self.threshold != threshold_for(self.difficulty)
282        {
283            return Err("faucet challenge has an inconsistent difficulty".into());
284        }
285        Ok(())
286    }
287
288    fn preimage(&self, nonce: u64) -> String {
289        format!(
290            "{}\n{}\n{}\n{}\n{}\n{}\n{}\n{nonce}",
291            self.domain,
292            self.chain_id,
293            self.checkpoint_seq,
294            self.checkpoint_digest,
295            self.random_bytes,
296            self.faucet_address,
297            self.recipient,
298        )
299    }
300}

345/// Preserve a faucet HTTP failure for inspection through `error.downcast_ref::<FaucetError>()`.
346///
347/// Check `code` and `digest` before requesting another payout. An error does not necessarily mean
348/// that no funds were sent. An attached challenge does not authorize an automatic retry.
349#[derive(Debug, Clone)]
350#[non_exhaustive]
351pub struct FaucetError {
352    pub status: StatusCode,
353    /// Preserve the server's error code, including unknown codes; `None` means absent or malformed.
354    pub code: Option<String>,
355    pub message: String,
356    /// Identify the payout transaction to inspect; `None` does not prove that no payout occurred.
357    pub digest: Option<Digest>,
358    /// Retain a decodable challenge supplied by the server, without validating it.
359    pub challenge: Option<PowChallenge>,
360}
361
362impl std::fmt::Display for FaucetError {
363    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
364        write!(f, "faucet request failed: {}", self.message)?;
365        if let Some(code) = &self.code {
366            write!(f, " ({code})")?;
367        }
368        Ok(())
369    }
370}

417fn deserialize_u64<'de, D>(deserializer: D) -> Result<u64, D::Error>
418where
419    D: de::Deserializer<'de>,
420{
421    let value = String::deserialize(deserializer)?;
422    if value.is_empty()
423        || (value.len() > 1 && value.starts_with('0'))
424        || !value.bytes().all(|byte| byte.is_ascii_digit())
425    {
426        return Err(de::Error::custom("expected a canonical decimal u64"));
427    }
428    value.parse().map_err(de::Error::custom)
429}

431fn hash_to_u64(hash: &[u8; POW_HASH_LENGTH]) -> u64 {
432    let mut value = [0; 8];
433    value.copy_from_slice(&hash[..8]);
434    u64::from_be_bytes(value)
435}

729    fn read_http_request(stream: &mut std::net::TcpStream) -> String {
730        let mut bytes = Vec::new();
731        let mut buffer = [0; 1024];
732        loop {
733            let size = stream.read(&mut buffer).unwrap();
734            bytes.extend_from_slice(&buffer[..size]);
735            let Some(headers_end) = bytes.windows(4).position(|window| window == b"\r\n\r\n")
736            else {
737                continue;
738            };
739            let headers = std::str::from_utf8(&bytes[..headers_end]).unwrap();
740            let content_length = headers
741                .lines()
742                .find_map(|header| header.strip_prefix("content-length: "))
743                .and_then(|value| value.parse::<usize>().ok())
744                .unwrap_or_default();
745            if bytes.len() >= headers_end + 4 + content_length {
746                return String::from_utf8(bytes).unwrap();
747            }
748        }
749    }

756        stream: &mut std::net::TcpStream,
757        status: u16,
758        reason: &str,
759        body: &str,
760    ) {
761        write!(
762            stream,
763            "HTTP/1.1 {status} {reason}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",

765        )
766        .unwrap();
767    }
