34/// Permitted `CheckpointBound` and `CursorBound` endings, with or without prior progress.
35#[derive(Clone, Copy)]
36pub(super) struct ExpectedEnd {
37    checkpoint_after_progress: bool,
38    cursor_after_progress: bool,
39    checkpoint_without_prior_progress: bool,
40    cursor_without_prior_progress: bool,
41}
42
43impl ExpectedEnd {
44    fn from_bounds(checkpoint: bool, cursor: bool) -> Self {
45        Self {
46            checkpoint_after_progress: checkpoint,
47            cursor_after_progress: cursor,
48            checkpoint_without_prior_progress: checkpoint,
49            cursor_without_prior_progress: cursor,
50        }
51    }
52
53    fn accepts(
54        self,
55        reason: QueryEndReason,
56        received_prior_frame: bool,
57        request_has_resume_cursor: bool,
58    ) -> bool {
59        let (checkpoint, cursor) = if received_prior_frame {
60            (self.checkpoint_after_progress, self.cursor_after_progress)
61        } else {
62            // A resume frontier may immediately return CursorBound when nothing follows it.
63            (
64                self.checkpoint_without_prior_progress,
65                self.cursor_without_prior_progress || request_has_resume_cursor,
66            )
67        };
68        (checkpoint && reason == QueryEndReason::CheckpointBound)
69            || (cursor && reason == QueryEndReason::CursorBound)
70    }
71}

73fn validate_query_end_item(reason: QueryEndReason, item_present: bool) -> Result<()> {
74    if reason == QueryEndReason::ItemLimit && !item_present {
75        Err(Status::data_loss("ItemLimit QueryEnd is missing its item"))
76    } else if reason != QueryEndReason::ItemLimit && item_present {
77        Err(Status::data_loss(
78            "non-ItemLimit QueryEnd unexpectedly contains an item",
79        ))
80    } else {
81        Ok(())
82    }
83}

85/// The boundary before the first checkpoint past `checkpoint_height`.
86fn past_checkpoint_height(checkpoint_height: u64) -> Result<Position> {
87    checkpoint_height
88        .checked_add(1)
89        .map(Position::checkpoint_start)
90        .ok_or_else(|| {
91            Status::out_of_range("checkpoint height cannot be converted to an exclusive end bound")
92        })
93}

95/// Lists from the template's start through `checkpoint_height`.
96pub(super) fn build_initial_list_request<A: SubscriptionAdapter>(
97    template: &A::ListRequest,
98    checkpoint_height: u64,
99) -> Result<(A::ListRequest, ExpectedEnd)> {
100    let mut request = template.clone();
101    set_end::<A>(

149/// Lists exactly the interval `gap` names.
150pub(super) fn build_recovery_list_request<A: SubscriptionAdapter>(
151    template: A::ListRequest,
152    gap: &RecoveryGap<A::Cursor>,
153) -> Result<(A::ListRequest, ExpectedEnd)> {
154    let mut request = template;
155    set_start::<A>(&mut request, gap.start, A::cursor_bytes(&gap.after));
156    let before = A::cursor_bytes(&gap.before);
157    let ends_at_cursor = before.is_some();
158    set_end::<A>(&mut request, gap.end, before)?;
159    // The end fields and the `before` cursor name the same point, so either may end the scan.
160    Ok((request, ExpectedEnd::from_bounds(true, ends_at_cursor)))
161}

163pub(super) fn determine_expected_end<A: SubscriptionAdapter>(
164    request: &A::ListRequest,
165    direction: ListScanDirection,
166) -> ExpectedEnd {
167    let options = A::options(request);
168    let checkpoint_without_prior_progress = A::start_checkpoint(request).is_some()
169        || A::end_checkpoint(request).is_some()
170        || direction == ListScanDirection::Descending;
171    let cursor_without_prior_progress =
172        options.is_some_and(|options| options.after.is_some() || options.before.is_some());
173    let (checkpoint_after_progress, cursor_after_progress) = match direction {
174        ListScanDirection::Ascending => (
175            A::end_checkpoint(request).is_some(),
176            options.is_some_and(|options| options.before.is_some()),
177        ),
178        ListScanDirection::Descending => {
179            (true, options.is_some_and(|options| options.after.is_some()))
180        }
181    };
182    ExpectedEnd {
183        checkpoint_after_progress,
184        cursor_after_progress,
185        checkpoint_without_prior_progress,
186        cursor_without_prior_progress,
187    }
188}

252    pub(super) fn new(
253        client: Client,
254        payload: A::ListRequest,
255        expected_end: ExpectedEnd,
256        direction: ListScanDirection,
257        ledger_tip_policy: LedgerTipPolicy,
258        observability: LedgerStreamObservability,
259        stage: LedgerStreamStage,
260    ) -> Self {
261        let mut machine = Self {
262            client,
263            payload,
264            expected_end,
265            direction,
266            ledger_tip_policy,
267            rpc_state: RpcState::Idle,
268            request_resume_position: None,
269            request_has_resume_cursor: false,
270            latest_progress: None,
271            polling_attempt: 0,
272            received_frame_in_request: false,
273            request_has_checkpoint_coverage: false,
274            retry: RetryState::new(A::FAMILY, LedgerStreamOperation::List),
275            observability,
276            stage,
277        };
278        machine.start_dispatch();
279        machine
280    }

282    fn start_dispatch(&mut self) {
283        self.received_frame_in_request = false;
284        self.request_has_checkpoint_coverage = false;
285        self.request_has_resume_cursor = A::options(&self.payload)
286            .is_some_and(|options| self.direction.has_resume_cursor(options));
287        let request = Request::new(self.payload.clone());
288        self.rpc_state = RpcState::Dispatch {
289            future: A::dispatch_list(self.client.clone(), request),
290            started_at: None,
291        };
292    }
293
294    /// Points the next request at the most recent frame's resume point.
295    fn resume_from_latest_progress(&mut self) -> Result<()> {
296        if let Some(progress) = &self.latest_progress {
297            self.direction
298                .set_resume::<A>(&mut self.payload, progress)?;
299            self.request_resume_position = Some(progress.position);
300        }
301        Ok(())
302    }
303
304    fn sleep(&mut self, delay: Duration) {
305        self.rpc_state = RpcState::Sleep(Box::pin(tokio::time::sleep(delay)));
306    }

308    /// Dropping this future preserves `self.rpc_state`, so the next call resumes rather than
309    /// restarts in-flight work.
310    pub(super) async fn poll_rpc(&mut self) -> RpcEvent<A::ListResponse> {
311        loop {
312            match &mut self.rpc_state {
313                RpcState::Idle => {
314                    return RpcEvent::Status(
315                        Status::internal("idle RPC state polled"),
316                        FailurePhase::Dispatch,
317                    );
318                }
319                RpcState::Dispatch { future, started_at } => {
320                    // The timer starts on the first poll and survives cancellation by gap buffering.
321                    if started_at.is_none() {
322                        *started_at = self.observability.start_timer();
323                    }
324                    let rpc_started_at = started_at.as_ref().cloned();
325                    let result = future.await;
326                    self.observability.emit_rpc_response(
327                        rpc_started_at,
328                        A::FAMILY,
329                        LedgerStreamOperation::List,
330                        self.stage,
331                        &result,
332                    );
333                    match result {
334                        Ok(stream) => self.rpc_state = RpcState::Stream(stream),
335                        Err(status) => {
336                            self.rpc_state = RpcState::Idle;
337                            return RpcEvent::Status(status, FailurePhase::Dispatch);
338                        }
339                    }
340                }
341                RpcState::Stream(stream) => match stream.next().await {
342                    Some(Ok(response)) => return RpcEvent::Frame(response),
343                    Some(Err(status)) => {
344                        self.rpc_state = RpcState::Idle;
345                        return RpcEvent::Status(status, FailurePhase::Body);
346                    }
347                    None => {
348                        self.rpc_state = RpcState::Idle;
349                        return RpcEvent::Eof;
350                    }
351                },
352                RpcState::Sleep(sleep) => {
353                    sleep.as_mut().await;
354                    self.rpc_state = RpcState::Idle;
355                    return RpcEvent::Wake;
356                }
357            }
358        }
359    }

361    /// Advances validation and pagination state with one RPC event.
362    pub(super) fn process_event(
363        &mut self,
364        event: RpcEvent<A::ListResponse>,
365        config: &LedgerStreamConfig,
366    ) -> ListAction<A::ListResponse> {
367        match event {
368            RpcEvent::Wake => {
369                self.start_dispatch();
370                ListAction::Continue
371            }
372            // Successful List streams end with QueryEnd; bare EOF has no completion boundary.
373            RpcEvent::Eof => ListAction::Terminal(Status::data_loss(
374                "List stream ended before its QueryEnd frame",
375            )),
376            RpcEvent::Status(status, phase) => match self.process_status(status, phase, config) {
377                Ok(()) => ListAction::Continue,
378                Err(status) => ListAction::Terminal(status),
379            },
380            RpcEvent::Frame(response) => {
381                let (item_present, watermark, end) = A::extract_metadata(&response);
382                match self.process_frame_metadata(item_present, watermark, end, config) {
383                    Ok((progress, complete)) => ListAction::Frame {
384                        response,
385                        progress,
386                        complete,
387                    },
388                    Err(status) => ListAction::Terminal(status),
389                }
390            }
391        }
392    }
393
394    fn process_status(
395        &mut self,
396        status: Status,
397        phase: FailurePhase,
398        config: &LedgerStreamConfig,
399    ) -> Result<()> {
400        self.resume_from_latest_progress()?;
401        // A request that resumes from a frame has bounds a server reported.
402        let bounds = if self.request_resume_position.is_some() {
403            RequestBounds::ServerReported
404        } else {
405            RequestBounds::CallerSupplied
406        };
407        if let Some(delay) = self.retry.retry_delay(
408            &status,
409            phase,
410            bounds,
411            config,
412            &self.observability,
413            self.stage,
414        ) {
415            self.sleep(delay);
416            Ok(())
417        } else {
418            Err(status)
419        }
420    }

422    fn process_frame_metadata(
423        &mut self,
424        item_present: bool,
425        watermark: Option<&Watermark>,
426        end: Option<&QueryEnd>,
427        config: &LedgerStreamConfig,
428    ) -> Result<(Option<Progress<Bytes>>, bool)> {
429        let watermark =
430            watermark.ok_or_else(|| Status::data_loss("List frame is missing its watermark"))?;
431        let cursor = watermark
432            .cursor
433            .clone()
434            .ok_or_else(|| Status::data_loss("List watermark is missing its cursor"))?;
435        let end_reason = end
436            .map(|end| {
437                end.reason
438                    .and_then(|reason| QueryEndReason::try_from(reason).ok())
439                    .filter(|reason| *reason != QueryEndReason::Unknown)
440                    .ok_or_else(|| {
441                        Status::data_loss("List QueryEnd has a missing or unknown reason")
442                    })
443            })
444            .transpose()?;
445
446        let checkpoint = watermark.checkpoint;
447        if checkpoint.is_none() && self.request_has_checkpoint_coverage {
448            return Err(Status::data_loss(
449                "List checkpoint coverage became unavailable",
450            ));
451        }
452        if checkpoint.is_some() {
453            self.request_has_checkpoint_coverage = true;
454        }
455
456        let mut frame_progress = match watermark.cursor_position.as_ref() {
457            Some(position) => Some(Progress {
458                position: A::GRANULARITY.position(position),
459                cursor,
460                checkpoint,
461            }),
462            // A server leaves the position unset only on a page's final frame when the request
463            // names that point by an `after` or `before` cursor alone and the cursor was minted
464            // without a position, by a server that predates cursor positions. A caller's cursor on
465            // the far side of the scan (`before` ascending, `after` descending) stays bare on every
466            // request. A near-side cursor is bare only on the caller's first request:
467            // continuations pair it with position fields.
468            None if !item_present
469                && matches!(
470                    end_reason,
471                    Some(
472                        QueryEndReason::LedgerTip
473                            | QueryEndReason::CursorBound
474                            | QueryEndReason::CheckpointBound
475                    )
476                )
477                && A::options(&self.payload).is_some_and(|options| {
478                    self.direction.has_far_cursor(options)
479                        || (self.request_resume_position.is_none()
480                            && self.direction.has_resume_cursor(options))
481                }) =>
482            {
483                None
484            }
485            None => return Err(missing_cursor_position("List")),
486        };
487        let mut advanced = item_present;
488        if let Some(next) = &mut frame_progress {
489            if let Some(latest) = &mut self.latest_progress {
490                next.inherit_checkpoint_coverage(latest);
491                let (position_advanced, _) =
492                    latest.advance(next, self.direction, item_present, "List")?;
493                advanced |= position_advanced;
494            } else {
495                advanced = true;
496                self.latest_progress = Some(next.clone());
497            }
498        }
499        let received_prior_frame_in_request = self.received_frame_in_request;
500        self.received_frame_in_request = true;
501        if advanced {
502            self.polling_attempt = 0;
503            self.retry.reset(&self.observability);
504        }
505
506        let complete = match end_reason {
507            None => false,
508            Some(reason) => {
509                validate_query_end_item(reason, item_present)?;
510                if reason == QueryEndReason::CheckpointBound {
511                    A::validate_checkpoint_bound(&self.payload, self.direction, checkpoint)?;
512                }
513
514                match reason {
515                    QueryEndReason::ItemLimit | QueryEndReason::ScanLimit => {
516                        // An ItemLimit item already advanced past the previous frame.
517                        if reason == QueryEndReason::ScanLimit
518                            && let (Some(progress), Some(resumed_from)) =
519                                (&frame_progress, self.request_resume_position)
520                            && !self.direction.beyond(progress.position, resumed_from)
521                        {
522                            return Err(Status::data_loss(
523                                "ScanLimit QueryEnd did not advance its cursor_position",
524                            ));
525                        }
526                        self.resume_from_latest_progress()?;
527                        self.start_dispatch();
528                        false
529                    }
530                    QueryEndReason::CheckpointBound | QueryEndReason::CursorBound => {
531                        if !self.expected_end.accepts(
532                            reason,
533                            received_prior_frame_in_request,
534                            self.request_has_resume_cursor,
535                        ) {
536                            return Err(Status::data_loss("List ended at an unexpected bound"));
537                        }
538                        self.rpc_state = RpcState::Idle;
539                        true
540                    }
541                    // Servers end only ascending scans at the ledger tip.
542                    QueryEndReason::LedgerTip
543                        if self.direction == ListScanDirection::Descending =>
544                    {
545                        return Err(Status::data_loss("descending List reached LedgerTip"));
546                    }
547                    QueryEndReason::LedgerTip
548                        if matches!(self.ledger_tip_policy, LedgerTipPolicy::Complete) =>
549                    {
550                        self.rpc_state = RpcState::Idle;
551                        true
552                    }
553                    QueryEndReason::LedgerTip => {
554                        self.resume_from_latest_progress()?;
555                        let delay = backoff_delay(config, self.polling_attempt);
556                        self.polling_attempt = self.polling_attempt.saturating_add(1);
557                        self.sleep(delay);
558                        false
559                    }
560                    QueryEndReason::Unknown => {
561                        return Err(Status::data_loss("List QueryEnd has an unknown reason"));
562                    }
563                }
564            }
565        };
566
567        Ok((frame_progress, complete))
568    }
569}

571/// Drives a finite List operation by resuming the request cursor and yielding whole responses.
572pub(super) struct ListDriver<A: SubscriptionAdapter> {
573    machine: Option<ListMachine<A>>,
574    config: LedgerStreamConfig,
575    observability: LedgerStreamObservability,
576    terminal: Option<Status>,
577    done: bool,
578}
579
580impl<A: SubscriptionAdapter> ListDriver<A> {
581    pub(super) fn new(client: Client, request: A::ListRequest, config: LedgerStreamConfig) -> Self {
582        let observability = LedgerStreamObservability::new(config.observer());
583        let payload = request;
584        let direction = ListScanDirection::from_request::<A>(&payload);
585        let terminal = direction.as_ref().err().cloned();
586        let machine = direction.ok().map(|direction| {
587            let expected_end = determine_expected_end::<A>(&payload, direction);
588            ListMachine::new(
589                client,
590                payload,
591                expected_end,
592                direction,
593                LedgerTipPolicy::Complete,
594                observability.clone(),
595                LedgerStreamStage::List,
596            )
597        });
598        Self {
599            machine,
600            config,
601            observability,
602            terminal,
603            done: false,
604        }
605    }
606
607    pub(super) async fn next(&mut self) -> Option<Result<A::ListResponse>> {
608        if self.done {
609            return None;
610        }
611        if let Some(status) = self.terminal.take() {
612            return Some(self.finish_terminal(status));
613        }
614
615        loop {
616            let machine = self.machine.as_mut()?;
617            let event = machine.poll_rpc().await;
618            match machine.process_event(event, &self.config) {
619                ListAction::Frame {
620                    response, complete, ..
621                } => {
622                    if complete {
623                        self.done = true;
624                        self.machine = None;
625                    }
626                    return Some(Ok(response));
627                }
628                ListAction::Continue => {}
629                ListAction::Terminal(status) => return Some(self.finish_terminal(status)),
630            }
631        }
632    }
633
634    fn finish_terminal(&mut self, status: Status) -> Result<A::ListResponse> {
635        self.observability
636            .emit(|| LedgerStreamEvent::TerminalError {
637                family: A::FAMILY,
638                status: status.clone(),
639            });
640        self.done = true;
641        self.machine = None;
642        Err(status)
643    }
644}
