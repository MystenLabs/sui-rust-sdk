31/// Permitted `CheckpointBound` and `CursorBound` endings, with or without prior progress.
32#[derive(Clone, Copy)]
33pub(super) struct ExpectedEnd {
34    checkpoint_after_progress: bool,
35    cursor_after_progress: bool,
36    checkpoint_without_prior_progress: bool,
37    cursor_without_prior_progress: bool,
38}
39
40impl ExpectedEnd {
41    fn from_bounds(checkpoint: bool, cursor: bool) -> Self {
42        Self {
43            checkpoint_after_progress: checkpoint,
44            cursor_after_progress: cursor,
45            checkpoint_without_prior_progress: checkpoint,
46            cursor_without_prior_progress: cursor,
47        }
48    }
49
50    fn accepts(
51        self,
52        reason: QueryEndReason,
53        received_prior_frame: bool,
54        request_started_from_resume_bound: bool,
55    ) -> bool {
56        let (checkpoint, cursor) = if received_prior_frame {
57            (self.checkpoint_after_progress, self.cursor_after_progress)
58        } else {
59            // A resume frontier may immediately return CursorBound when nothing follows it.
60            (
61                self.checkpoint_without_prior_progress,
62                self.cursor_without_prior_progress || request_started_from_resume_bound,
63            )
64        };
65        (checkpoint && reason == QueryEndReason::CheckpointBound)
66            || (cursor && reason == QueryEndReason::CursorBound)
67    }
68}

70fn validate_query_end_item(reason: QueryEndReason, item_present: bool) -> Result<()> {
71    if reason == QueryEndReason::ItemLimit && !item_present {
72        Err(Status::data_loss("ItemLimit QueryEnd is missing its item"))
73    } else if reason != QueryEndReason::ItemLimit && item_present {
74        Err(Status::data_loss(
75            "non-ItemLimit QueryEnd unexpectedly contains an item",
76        ))
77    } else {
78        Ok(())
79    }
80}

82fn to_exclusive_end_bound(checkpoint_height: u64) -> Result<u64> {
83    checkpoint_height.checked_add(1).ok_or_else(|| {
84        Status::out_of_range("checkpoint height cannot be converted to an exclusive end bound")
85    })
86}

96pub(super) fn build_initial_list_request<A: SubscriptionAdapter>(
97    template: &A::ListRequest,
98    checkpoint_height: u64,
99    list_page_limit: Option<u32>,
100) -> Result<(A::ListRequest, ExpectedEnd)> {
101    let mut request = template.clone();
102    apply_internal_list_page_limit::<A>(&mut request, list_page_limit);
103    A::set_end_checkpoint(
104        &mut request,
105        Some(to_exclusive_end_bound(checkpoint_height)?),
106    );
107    Ok((request, ExpectedEnd::from_bounds(true, false)))
108}

110/// Queries a single tip checkpoint to establish baseline watermark progress (used when starting in

147/// Lists strictly after `committed_progress` through the server's indexed tip, with no end bound.
148/// The scan's last frame claims the tip it reached, or no checkpoint when `committed_progress`
149/// lies past that tip.
150pub(super) fn build_resume_to_tip_list_request<A: SubscriptionAdapter>(
151    template: &A::ListRequest,
152    committed_progress: &Progress<A::Cursor>,
153    list_page_limit: Option<u32>,
154) -> Result<(A::ListRequest, ExpectedEnd)> {
155    let mut request = template.clone();
156    apply_internal_list_page_limit::<A>(&mut request, list_page_limit);
157    A::set_ascending_resume(&mut request, committed_progress)?;
158    A::options_mut(&mut request).before = None;
159    A::set_end_checkpoint(&mut request, None);
160    let expected_end = determine_expected_end::<A>(&request, ListScanDirection::Ascending);
161    Ok((request, expected_end))

164pub(super) fn build_recovery_list_request<A: SubscriptionAdapter>(
165    template: A::ListRequest,
166    gap: &RecoveryGap,
167    list_page_limit: Option<u32>,
168) -> Result<(A::ListRequest, ExpectedEnd)> {
169    let mut request = template;
170    apply_internal_list_page_limit::<A>(&mut request, list_page_limit);
171    match gap {
172        RecoveryGap::Checkpoints {
173            start_checkpoint,
174            end_checkpoint,
175        } => {
176            A::set_start_checkpoint(&mut request, Some(*start_checkpoint));
177            A::set_end_checkpoint(&mut request, Some(*end_checkpoint));
178            let options = A::options_mut(&mut request);
179            options.after = None;
180            options.before = None;
181            Ok((request, ExpectedEnd::from_bounds(true, false)))
182        }
183        RecoveryGap::Cursors {
184            after,
185            upper: CursorGapUpper::Before(before),
186        } => {
187            let options = A::options_mut(&mut request);
188            options.after = Some(after.clone());
189            options.before = Some(before.clone());
190            Ok((request, ExpectedEnd::from_bounds(false, true)))
191        }
192        RecoveryGap::Cursors {
193            after,
194            upper: CursorGapUpper::EndOfCheckpoint(checkpoint),
195        } => {
196            let options = A::options_mut(&mut request);
197            options.after = Some(after.clone());
198            options.before = None;
199            A::set_end_checkpoint(&mut request, Some(to_exclusive_end_bound(*checkpoint)?));
200            Ok((request, ExpectedEnd::from_bounds(true, false)))
201        }
202    }
203}

205pub(super) fn determine_expected_end<A: SubscriptionAdapter>(
206    request: &A::ListRequest,
207    direction: ListScanDirection,
208) -> ExpectedEnd {
209    let options = A::options(request);
210    let checkpoint_without_prior_progress = A::start_checkpoint(request).is_some()
211        || A::end_checkpoint(request).is_some()
212        || direction == ListScanDirection::Descending;
213    let cursor_without_prior_progress =
214        options.is_some_and(|options| options.after.is_some() || options.before.is_some());
215    let (checkpoint_after_progress, cursor_after_progress) = match direction {
216        ListScanDirection::Ascending => (
217            A::end_checkpoint(request).is_some(),
218            options.is_some_and(|options| options.before.is_some()),
219        ),
220        ListScanDirection::Descending => {
221            (true, options.is_some_and(|options| options.after.is_some()))
222        }
223    };
224    ExpectedEnd {
225        checkpoint_after_progress,
226        cursor_after_progress,
227        checkpoint_without_prior_progress,
228        cursor_without_prior_progress,
229    }
230}

269/// How an ascending scan handles a `LedgerTip` end.
270#[derive(Clone, Copy)]
271pub(super) enum LedgerTipPolicy {
272    /// Complete at the indexed tip.
273    Complete,
274    /// Retry from the served frontier until the fixed bound is indexed.
275    WaitForExpectedBound,
276    /// Complete at a tip whose end frame claims checkpoint coverage. Retry from the served
277    /// frontier while the end frame claims none, as it does for an `after` cursor past the tip.
278    WaitForCoverage,
279}

335    fn start_dispatch(&mut self) {
336        self.received_frame_in_request = false;
337        self.request_has_checkpoint_coverage = false;
338        self.request_start_resume_bound = A::options(&self.payload)
339            .and_then(|options| self.direction.resume_bound(options).clone());
340        let request = Request::new(self.payload.clone());
341        self.rpc_state = RpcState::Dispatch {
342            future: A::dispatch_list(self.client.clone(), request),
343            started_at: None,
344        };
345    }

347    fn sleep(&mut self, delay: Duration) {
348        self.rpc_state = RpcState::Sleep(Box::pin(tokio::time::sleep(delay)));
349    }

351    /// Dropping this future preserves `self.rpc_state`, so the next call resumes rather than
352    /// restarts in-flight work.
353    pub(super) async fn poll_rpc(&mut self) -> RpcEvent<A::ListResponse> {
354        loop {
355            match &mut self.rpc_state {
356                RpcState::Idle => {
357                    return RpcEvent::Status(
358                        Status::internal("idle RPC state polled"),
359                        FailurePhase::Dispatch,
360                    );
361                }
362                RpcState::Dispatch { future, started_at } => {
363                    // The timer starts on the first poll and survives cancellation by gap buffering.
364                    if started_at.is_none() {
365                        *started_at = self.observability.start_timer();
366                    }
367                    let rpc_started_at = started_at.as_ref().cloned();
368                    let result = future.await;
369                    self.observability.emit_rpc_response(
370                        rpc_started_at,
371                        A::FAMILY,
372                        LedgerStreamOperation::List,
373                        self.stage,
374                        &result,
375                    );
376                    match result {
377                        Ok(stream) => self.rpc_state = RpcState::Stream(stream),
378                        Err(status) => {
379                            self.rpc_state = RpcState::Idle;
380                            return RpcEvent::Status(status, FailurePhase::Dispatch);
381                        }
382                    }
383                }
384                RpcState::Stream(stream) => match stream.next().await {
385                    Some(Ok(response)) => return RpcEvent::Frame(response),
386                    Some(Err(status)) => {
387                        self.rpc_state = RpcState::Idle;
388                        return RpcEvent::Status(status, FailurePhase::Body);
389                    }
390                    None => {
391                        self.rpc_state = RpcState::Idle;
392                        return RpcEvent::Eof;
393                    }
394                },
395                RpcState::Sleep(sleep) => {
396                    sleep.as_mut().await;
397                    self.rpc_state = RpcState::Idle;
398                    return RpcEvent::Wake;
399                }
400            }
401        }
402    }

404    /// Advances validation and pagination state with one RPC event.
405    pub(super) fn process_event(
406        &mut self,
407        event: RpcEvent<A::ListResponse>,
408        config: &LedgerStreamConfig,
409    ) -> ListAction<A::ListResponse, Progress<A::Cursor>> {
410        match event {
411            RpcEvent::Wake => {
412                self.start_dispatch();
413                ListAction::Continue
414            }
415            // Successful List streams end with QueryEnd; bare EOF has no completion boundary.
416            RpcEvent::Eof => ListAction::Terminal(Status::data_loss(
417                "List stream ended before its QueryEnd frame",
418            )),
419            RpcEvent::Status(status, phase) => match self.process_status(status, phase, config) {
420                Ok(()) => ListAction::Continue,
421                Err(status) => ListAction::Terminal(status),
422            },
423            RpcEvent::Frame(response) => {
424                let (item_present, watermark, end) = A::extract_metadata(&response);
425                match self.process_frame_metadata(item_present, watermark, end, config) {
426                    Ok((progress, complete)) => ListAction::Frame {
427                        response,
428                        progress,
429                        complete,
430                    },
431                    Err(status) => ListAction::Terminal(status),
432                }
433            }
434        }
435    }
436
437    fn process_status(
438        &mut self,
439        status: Status,
440        phase: FailurePhase,
441        config: &LedgerStreamConfig,
442    ) -> Result<()> {
443        if let Some(cursor) = self.latest_cursor.clone() {
444            set_resume_bound::<A>(self.direction, &mut self.payload, cursor);
445        }
446        if let Some(delay) =
447            self.retry
448                .retry_delay(&status, phase, config, &self.observability, self.stage)
449        {
450            self.sleep(delay);
451            Ok(())
452        } else {
453            Err(status)
454        }
455    }
456
457    fn process_frame_metadata(
458        &mut self,
459        item_present: bool,
460        watermark: Option<&Watermark>,
461        end: Option<&QueryEnd>,
462        config: &LedgerStreamConfig,
463    ) -> Result<(Option<Progress<A::Cursor>>, bool)> {
464        let watermark =
465            watermark.ok_or_else(|| Status::data_loss("List frame is missing its watermark"))?;
466        let cursor = watermark
467            .cursor
468            .as_ref()
469            .cloned()
470            .ok_or_else(|| Status::data_loss("List watermark is missing its cursor"))?;
471
472        let checkpoint = watermark.checkpoint;
473        if checkpoint.is_none() && self.request_has_checkpoint_coverage {
474            return Err(Status::data_loss(
475                "List checkpoint coverage became unavailable",
476            ));
477        }
478        if checkpoint.is_some() {
479            self.request_has_checkpoint_coverage = true;
480        }
481        let received_prior_frame_in_request = self.received_frame_in_request;
482        let received_prior_frame_ever = self.received_any_frame;
483        if item_present
484            && end.is_none()
485            && !received_prior_frame_in_request
486            && self.request_start_resume_bound.as_ref() == Some(&cursor)
487        {
488            return Err(Status::data_loss(
489                "List item cursor equals its exclusive request resume bound",
490            ));
491        }
492        let mut frame_progress = A::Cursor::position(&cursor, checkpoint);
493        if frame_progress.is_none() && item_present {
494            return Err(Status::data_loss(
495                "List item watermark names no resumable position",
496            ));
497        }
498        if let (Some(previous), Some(next)) = (&self.latest_progress, &mut frame_progress) {
499            next.inherit_checkpoint_coverage(previous);
500            previous.validate_list_successor(next, self.direction)?;
501        }
502        self.received_frame_in_request = true;
503        self.received_any_frame = true;
504        let cursor_changed = self.latest_cursor.as_ref() != Some(&cursor);
505        if item_present && !cursor_changed {
506            return Err(Status::data_loss("List item repeated its cursor"));
507        }
508        if cursor_changed {
509            self.latest_cursor = Some(cursor.clone());
510            self.polling_attempt = 0;
511            self.retry.reset(&self.observability);
512        }
513        if let Some(progress) = &frame_progress {
514            self.latest_progress = Some(progress.clone());
515        }
516
517        let complete = match end {
518            None => false,
519            Some(end) => {
520                let reason = end
521                    .reason
522                    .and_then(|reason| QueryEndReason::try_from(reason).ok())
523                    .filter(|reason| *reason != QueryEndReason::Unknown)
524                    .ok_or_else(|| {
525                        Status::data_loss("List QueryEnd has a missing or unknown reason")
526                    })?;
527                validate_query_end_item(reason, item_present)?;
528                if reason == QueryEndReason::CheckpointBound {
529                    A::validate_checkpoint_bound(&self.payload, self.direction, checkpoint)?;
530                }
531
532                match reason {
533                    QueryEndReason::ItemLimit => {
534                        if self.request_start_resume_bound.as_ref() == Some(&cursor) {
535                            return Err(Status::data_loss(
536                                "ItemLimit QueryEnd did not advance its cursor",
537                            ));
538                        }
539                        set_resume_bound::<A>(self.direction, &mut self.payload, cursor.clone());
540                        self.start_dispatch();
541                        false
542                    }
543                    QueryEndReason::ScanLimit => {
544                        if self.request_start_resume_bound.as_ref() == Some(&cursor) {
545                            return Err(Status::data_loss(
546                                "ScanLimit QueryEnd did not advance its cursor",
547                            ));
548                        }
549                        set_resume_bound::<A>(self.direction, &mut self.payload, cursor.clone());
550                        self.start_dispatch();
551                        false
552                    }
553                    QueryEndReason::CheckpointBound | QueryEndReason::CursorBound => {
554                        if !self.expected_end.accepts(
555                            reason,
556                            received_prior_frame_in_request,
557                            self.request_start_resume_bound.is_some(),
558                        ) {
559                            return Err(Status::data_loss("List ended at an unexpected bound"));
560                        }
561                        self.rpc_state = RpcState::Idle;
562                        true
563                    }
564                    // A descending interval entirely beyond the indexed tip ends at LedgerTip.
565                    QueryEndReason::LedgerTip
566                        if self.direction == ListScanDirection::Descending
567                            && !received_prior_frame_ever =>
568                    {
569                        self.rpc_state = RpcState::Idle;
570                        true
571                    }
572                    // After progress, a descending scan cannot legitimately reach LedgerTip.
573                    QueryEndReason::LedgerTip
574                        if self.direction == ListScanDirection::Descending =>
575                    {
576                        return Err(Status::data_loss(
577                            "descending List reached LedgerTip after prior scan progress",
578                        ));
579                    }
580                    QueryEndReason::LedgerTip
581                        if matches!(self.ledger_tip_policy, LedgerTipPolicy::Complete)
582                            || matches!(
583                                self.ledger_tip_policy,
584                                LedgerTipPolicy::WaitForCoverage
585                            ) && checkpoint.is_some() =>
586                    {
587                        self.rpc_state = RpcState::Idle;
588                        true
589                    }
590                    QueryEndReason::LedgerTip => {
591                        set_resume_bound::<A>(self.direction, &mut self.payload, cursor.clone());
592                        let delay = backoff_delay(config, self.polling_attempt);
593                        self.polling_attempt = self.polling_attempt.saturating_add(1);
594                        self.sleep(delay);
595                        false
596                    }
597                    QueryEndReason::Unknown => {
598                        return Err(Status::data_loss("List QueryEnd has an unknown reason"));
599                    }
600                }
601            }
602        };
603
604        Ok((frame_progress, complete))
605    }
606}

608/// Drives a finite List operation by resuming the request cursor and yielding whole responses.
609pub(super) struct ListDriver<A: SubscriptionAdapter> {
610    machine: Option<ListMachine<A>>,
611    config: LedgerStreamConfig,
612    observability: LedgerStreamObservability,
613    terminal: Option<Status>,
614    done: bool,
615}
616
617impl<A: SubscriptionAdapter> ListDriver<A> {
618    pub(super) fn new(client: Client, request: A::ListRequest, config: LedgerStreamConfig) -> Self {
619        let observability = LedgerStreamObservability::new(config.observer());
620        let payload = request;
621        let direction = ListScanDirection::from_request::<A>(&payload);
622        let terminal = direction.as_ref().err().cloned();
623        let machine = direction.ok().map(|direction| {
624            let expected_end = determine_expected_end::<A>(&payload, direction);
625            ListMachine::new(
626                client,
627                payload,
628                expected_end,
629                direction,
630                LedgerTipPolicy::Complete,
631                observability.clone(),
632                LedgerStreamStage::List,
633            )
634        });
635        Self {
636            machine,
637            config,
638            observability,
639            terminal,
640            done: false,
641        }
642    }
643
644    pub(super) async fn next(&mut self) -> Option<Result<A::ListResponse>> {
645        if self.done {
646            return None;
647        }
648        if let Some(status) = self.terminal.take() {
649            return Some(self.finish_terminal(status));
650        }
651
652        loop {
653            let machine = self.machine.as_mut()?;
654            let event = machine.poll_rpc().await;
655            match machine.process_event(event, &self.config) {
656                ListAction::Frame {
657                    response, complete, ..
658                } => {
659                    if complete {
660                        self.done = true;
661                        self.machine = None;
662                    }
663                    return Some(Ok(response));
664                }
665                ListAction::Continue => {}
666                ListAction::Terminal(status) => return Some(self.finish_terminal(status)),
667            }
668        }
669    }
670
671    fn finish_terminal(&mut self, status: Status) -> Result<A::ListResponse> {
672        self.observability
673            .emit(|| LedgerStreamEvent::TerminalError {
674                family: A::FAMILY,
675                status: status.clone(),
676            });
677        self.done = true;
678        self.machine = None;
679        Err(status)
680    }
